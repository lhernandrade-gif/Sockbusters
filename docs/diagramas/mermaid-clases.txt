        +string passwordHash
        +string rol

        +iniciarSesion()
        +cerrarSesion()
    }

        +registrarProducto()
        +actualizarProducto()
        +actualizarStock()
        +modificarPrecio()
        +gestionarColeccion()
    }

    class TokenJWT {
        +string token
        +datetime fechaEmision
        +datetime fechaExpiracion
        +boolean valido
        +validar()
        +expirar()
    }

    Usuario <|-- Administrador
    Usuario "1" --> "0..*" TokenJWT : obtiene

        +eliminarProducto()
    }

    class Obra {
        +int id
        +string titulo
        +string tipo
        +string descripcion
    }

    class Talla {
        +int id
        +string nombre
    }

    class Stock {
        +int id
        +int cantidadDisponible
        +int stockMinimo
        +consultarDisponibilidad()
        +aumentarStock()
        +disminuirStock()
    }

    Producto "0..*" --> "1" Categoria : pertenece a

    class Carrito {
        +int id
        +datetime fechaCreacion
        +calcularTotal()
        +agregarProducto()
        +modificarCantidad()
        +eliminarProducto()

        +calcularTotal()
        +actualizarEstado()
    }

    class ItemPedido {
        +int cantidad
        +decimal precioUnitario
        +calcularSubtotal()
    }

    class Checkout {
        +int id
        +decimal total
        +string estado
        +iniciar()
        +procesarPago()
        +confirmarPedido()
    }

    class Pago {
        +int id
        +decimal monto
        +datetime fecha
        +string estado
        +string metodo
        +procesar()
        +confirmar()
    }

    Usuario "1" --> "0..*" Pedido : realiza
    Pedido "1" *-- "1..*" ItemPedido : contiene
    ItemPedido "*" --> "1" Producto : corresponde a
    ItemPedido "*" --> "1" Talla : selecciona

    Carrito "1" --> "0..1" Pedido : genera

    Checkout "1" --> "1" Pago : genera
